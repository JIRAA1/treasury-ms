## 1. สรุปปัญหาปัจจุบัน

### 1.1 Root Cause

ระบบปัจจุบันใช้ `week: integer` เป็น Primary Key ของ `week_settings` และเป็น Foreign Key ใน `payments` และ `payment_credits` ส่งผลให้:

| ข้อจำกัด | ผลกระทบ |
|---|---|
| `week` เป็น integer ตายตัว (1–20) | เพิ่มงวดเกิน 20 ไม่ได้ |
| ชื่องวด hard-code เป็น "Week X" | เปลี่ยนเป็น "เดือน มิ.ย." หรืออื่นๆ ไม่ได้ |
| ไม่มีแนวคิด "เทอม" (Semester) | ข้อมูลหลายเทอมปนกัน, Reset = ลบทิ้งหมด |
| `TOTAL_WEEKS = 12` hard-code ใน dashboard | จำนวนงวดต่อเทอมเปลี่ยนไม่ได้จาก UI |
| `payment_credits` ผูกกับ `week` เช่นกัน | ต้องแก้พร้อมกัน 3 ตาราง |

### 1.2 ตารางที่ได้รับผลกระทบ

```
week_settings      ← จะถูกแทนที่ด้วย semesters + periods
payments           ← week: integer → period_id: uuid
payment_credits    ← week: integer → period_id: uuid
```

### 1.3 Files ที่ต้องแก้ (สรุป)

```
supabase_schema.sql           ← drop & recreate
src/types/index.ts            ← WeekStatus → PeriodStatus, Period, Semester
src/lib/utils.ts              ← ลบ getWeekLabel()
src/components/payments/
  WeekGrid.tsx                ← เปลี่ยนชื่อและ props
src/app/(student)/
  dashboard/page.tsx          ← ลบ TOTAL_WEEKS hardcode
  upload/page.tsx             ← week selector → period selector
  history/page.tsx            ← group by period.label
src/app/(admin)/
  overview/page.tsx           ← weekly bar chart → period bar chart
  payments/page.tsx           ← filter by period_id
  settings/page.tsx           ← เพิ่ม Period Manager UI
src/app/api/
  payments/upload/route.ts    ← รับ period_id แทน week
  payments/verify/route.ts    ← query by period_id
  payments/history/route.ts   ← filter by period_id
  reports/export/route.ts     ← groupBy period.label
```

## 2. ภาพรวม Schema ใหม่

### ความสัมพันธ์หลัก

```
semesters (1) ──< (many) periods (1) ──< (many) payments
                                    (1) ──< (many) payment_credits
```

### เปรียบเทียบ Before / After

| จุด | Before | After |
|---|---|---|
| งวดอ้างอิง | `week: integer` | `period_id: uuid` |
| ตั้งชื่องวด | ไม่ได้ (Week 1, Week 2) | ได้อิสระ เช่น "เดือน มิ.ย." |
| จำนวนงวดต่อเทอม | fixed ที่ 12 | กี่งวดก็ได้ |
| เก็บหลายเทอม | ไม่ได้ (Reset = ลบ) | ได้ผ่าน `semesters` |
| สลับ active เทอม | ไม่มี | `semesters.is_active` |

## 3. Phase 1 — Database: เพิ่ม Tables ใหม่

> **เป้าหมาย:** เพิ่ม `semesters`, `periods`, และ column `period_id` โดยยังไม่ลบของเดิม  
> **ความเสี่ยง:** ต่ำมาก — เป็นแค่ `CREATE TABLE` และ `ADD COLUMN`  
> **Rollback:** `DROP TABLE semesters, periods; ALTER TABLE payments DROP COLUMN period_id;`

```sql
-- ============================================================
-- Phase 1A: สร้าง semesters
-- ============================================================
CREATE TABLE public.semesters (
  id          uuid        NOT NULL DEFAULT gen_random_uuid(),
  name        text        NOT NULL,
  description text,
  is_active   boolean     NOT NULL DEFAULT false,
  created_at  timestamptz DEFAULT now(),
  CONSTRAINT semesters_pkey PRIMARY KEY (id)
);

-- Partial unique index: มีแค่ 1 active semester ในเวลาเดียวกัน
CREATE UNIQUE INDEX semesters_one_active_idx
  ON public.semesters (is_active)
  WHERE is_active = true;

-- ============================================================
-- Phase 1B: สร้าง periods (ทดแทน week_settings)
--           รักษา columns พิเศษจาก week_settings ทั้งหมด
-- ============================================================
CREATE TABLE public.periods (
  id                      uuid        NOT NULL DEFAULT gen_random_uuid(),
  semester_id             uuid        NOT NULL REFERENCES public.semesters(id) ON DELETE CASCADE,
  label                   text        NOT NULL,
  period_order            integer     NOT NULL,
  amount                  numeric     DEFAULT 100.00,
  base_amount             numeric     DEFAULT 50.00,
  late_fine_amount        numeric     DEFAULT 0.00,
  activity_type           text        CHECK (activity_type = ANY (ARRAY['small','medium','large'])),
  activity_extra_amount   numeric     DEFAULT 0,
  is_separate_collection  boolean     DEFAULT false,
  qr_url                  text,
  open_at                 timestamptz,
  close_at                timestamptz,
  deadline                timestamptz NOT NULL,
  created_at              timestamptz DEFAULT now(),
  CONSTRAINT periods_pkey PRIMARY KEY (id),
  CONSTRAINT periods_semester_order_unique UNIQUE (semester_id, period_order)
);

-- ============================================================
-- Phase 1C: เพิ่ม period_id ใน payments (nullable ก่อน)
-- ============================================================
ALTER TABLE public.payments
  ADD COLUMN period_id uuid REFERENCES public.periods(id);

-- ============================================================
-- Phase 1D: เพิ่ม period_id ใน payment_credits (nullable ก่อน)
-- ============================================================
ALTER TABLE public.payment_credits
  ADD COLUMN period_id uuid REFERENCES public.periods(id);

-- ============================================================
-- Phase 1E: RLS สำหรับตารางใหม่
-- ============================================================
ALTER TABLE public.semesters ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.periods   ENABLE ROW LEVEL SECURITY;

-- semesters: ทุกคนดูได้, admin เท่านั้นแก้ได้
CREATE POLICY "Anyone can view semesters" ON public.semesters
  FOR SELECT USING (true);

CREATE POLICY "Admins can manage semesters" ON public.semesters
  FOR ALL USING (
    EXISTS (
      SELECT 1 FROM public.users u
      WHERE u.id::text = auth.uid()::text
        AND u.role IN ('admin','treasurer')
    )
  );

-- periods: ทุกคนดูได้, admin เท่านั้นแก้ได้
CREATE POLICY "Anyone can view periods" ON public.periods
  FOR SELECT USING (true);

CREATE POLICY "Admins can manage periods" ON public.periods
  FOR ALL USING (
    EXISTS (
      SELECT 1 FROM public.users u
      WHERE u.id::text = auth.uid()::text
        AND u.role IN ('admin','treasurer')
    )
  );
```

## 4. Phase 2 — Database: Migrate ข้อมูลเดิม

> **เป้าหมาย:** Copy ข้อมูลจาก `week_settings` → `periods` และ backfill `period_id` ใน `payments`, `payment_credits`  
> **ข้อควรระวัง:** รัน Phase 1 ให้เสร็จก่อน, จด `<semester_id>` ที่ได้จาก INSERT  
> **Rollback:** ลบ rows ออกจาก `semesters` และ `periods`, ทำ `UPDATE payments SET period_id = NULL`

```sql
-- ============================================================
-- Phase 2A: สร้าง semester เริ่มต้น
--           ** จด uuid ที่ได้ไว้ใช้ใน Step 2B **
-- ============================================================
INSERT INTO public.semesters (name, description, is_active)
VALUES ('1/2567', 'นำเข้าจากระบบเก่า (week_settings)', true)
RETURNING id;

-- ============================================================
-- Phase 2B: Copy week_settings → periods
--           แทนที่ '<semester_id>' ด้วย uuid จาก Phase 2A
-- ============================================================
INSERT INTO public.periods (
  semester_id,
  label,
  period_order,
  amount,
  base_amount,
  late_fine_amount,
  activity_type,
  activity_extra_amount,
  is_separate_collection,
  qr_url,
  open_at,
  close_at,
  deadline
)
SELECT
  '<semester_id>',
  COALESCE(NULLIF(title, ''), 'สัปดาห์ ' || week::text),
  week,
  amount,
  base_amount,
  late_fine_amount,
  activity_type,
  activity_extra_amount,
  is_separate_collection,
  qr_url,
  payment_open_at,
  payment_close_at,
  deadline
FROM public.week_settings
ORDER BY week;

-- ============================================================
-- Phase 2C: Backfill period_id ใน payments
-- ============================================================
UPDATE public.payments p
SET period_id = pr.id
FROM public.periods pr
WHERE pr.period_order = p.week
  AND pr.semester_id  = '<semester_id>';

-- ============================================================
-- Phase 2D: Backfill period_id ใน payment_credits
-- ============================================================
UPDATE public.payment_credits pc
SET period_id = pr.id
FROM public.periods pr
WHERE pr.period_order = pc.week
  AND pr.semester_id  = '<semester_id>';

-- ============================================================
-- Phase 2E: ตรวจสอบ — ทั้งสองต้องได้ 0
-- ============================================================
SELECT 'payments orphaned'        AS check, COUNT(*) AS count
  FROM public.payments        WHERE period_id IS NULL AND week IS NOT NULL
UNION ALL
SELECT 'payment_credits orphaned' AS check, COUNT(*) AS count
  FROM public.payment_credits WHERE period_id IS NULL AND week IS NOT NULL;
```

## 5. Phase 3 — Database: Finalize & Cleanup

> **เป้าหมาย:** Drop ของเดิมออก, enforce constraints ใหม่  
> **เงื่อนไขก่อนรัน:** Frontend ทั้งหมดต้องใช้ `period_id` แล้ว 100%, ทดสอบ E2E ผ่านแล้ว  
> **Rollback:** ไม่มี — phase นี้ destructive, backup ก่อนเสมอ

```sql
-- ============================================================
-- Phase 3A: ทำ period_id เป็น NOT NULL
-- ============================================================
ALTER TABLE public.payments
  ALTER COLUMN period_id SET NOT NULL;

ALTER TABLE public.payment_credits
  ALTER COLUMN period_id SET NOT NULL;

-- ============================================================
-- Phase 3B: เพิ่ม UNIQUE constraint ใหม่ (แทน user+week)
-- ============================================================
ALTER TABLE public.payments
  ADD CONSTRAINT payments_user_period_unique UNIQUE (user_id, period_id);

-- ============================================================
-- Phase 3C: Drop week column และ FK เดิม
-- ============================================================

-- payments
ALTER TABLE public.payments
  DROP CONSTRAINT IF EXISTS payments_week_fkey,
  DROP COLUMN week;

-- payment_credits
ALTER TABLE public.payment_credits
  DROP CONSTRAINT IF EXISTS payment_credits_week_fkey,
  DROP COLUMN week;

-- ============================================================
-- Phase 3D: Backup แล้ว Drop week_settings
-- ============================================================
CREATE TABLE public.week_settings_backup
  AS SELECT * FROM public.week_settings;

DROP TABLE public.week_settings;

-- ============================================================
-- Phase 3E: อัปเดต SQL Helper Functions
-- ============================================================
DROP FUNCTION IF EXISTS get_week_collection_rate(integer);

CREATE OR REPLACE FUNCTION get_period_collection_rate(target_period_id uuid)
RETURNS numeric AS $$
  SELECT ROUND(
    COUNT(CASE WHEN status = 'approved' THEN 1 END)::numeric /
    NULLIF((SELECT COUNT(*) FROM public.users WHERE role = 'student'), 0) * 100,
    1
  )
  FROM public.payments
  WHERE period_id = target_period_id
$$ LANGUAGE SQL SECURITY DEFINER;

-- ============================================================
-- Phase 3F: อัปเดต get_treasury_balance (ไม่เปลี่ยน logic)
-- ============================================================
CREATE OR REPLACE FUNCTION get_treasury_balance()
RETURNS numeric AS $$
  SELECT
    COALESCE(SUM(CASE WHEN p.status = 'approved' THEN p.amount ELSE 0 END), 0) -
    COALESCE((SELECT SUM(e.amount) FROM public.expenses e WHERE e.approved_by IS NOT NULL), 0)
  FROM public.payments p
$$ LANGUAGE SQL SECURITY DEFINER;
```

const { data: semester } = await supabase
  .from('semesters')
  .select('*, periods(*)')
  .eq('is_active', true)
  .order('period_order', { referencedTable: 'periods', ascending: true })
  .single()

// Map payments เข้ากับ periods
const periodStatuses: PeriodStatus[] = (semester?.periods ?? []).map(period => {
  const payment = payments?.find(p => p.period_id === period.id)
  return {
    period,
    status: payment?.status === 'approved' ? 'paid'
          : payment?.status === 'pending'  ? 'pending'
          : 'unpaid',
    payment,
  }
})

// currentPeriod: งวดล่าสุดที่ยังเปิดรับอยู่

const currentPeriod = semester?.periods?.find(p =>
